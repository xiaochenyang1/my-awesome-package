## 🏗️ 整体架构设计

### 三层架构

```
┌─────────────────────────────────────────────────────────┐
│                    应用层 (App Layer)                    │
│          用户的 React 应用  |  用户的 Vue 应用            │
└─────────────────────────────────────────────────────────┘
                              ↓
┌─────────────────────────────────────────────────────────┐
│                 适配层 (Adapter Layer)                   │
│         @yourname/flow-react  |  @yourname/flow-vue     │
│         - React 组件封装       - Vue 组件封装            │
│         - React Hooks         - Vue Composables         │
└─────────────────────────────────────────────────────────┘
                              ↓
┌─────────────────────────────────────────────────────────┐
│                  核心层 (Core Layer)                     │
│              @yourname/flow-core                         │
│         - 数据模型 (Model)                               │
│         - 业务逻辑 (Logic)                               │
│         - 状态管理 (State)                               │
│         - 事件系统 (Event)                               │
│         - 配置引擎 (Config Engine)                       │
│         - 渲染引擎 (Render Engine)                       │
└─────────────────────────────────────────────────────────┘
```

### 核心设计思想

1. **配置驱动** 📋
   - 所有功能都通过 JSON 配置
   - 不写死任何逻辑在组件里
   - 支持动态加载配置

2. **框架无关** 🔄
   - 核心逻辑用纯 TypeScript/JavaScript
   - 不依赖任何框架
   - 适配层负责桥接框架

3. **插件化** 🔌
   - 节点类型可扩展
   - 表单组件可扩展
   - 渲染引擎可替换

// packages/core/src/models/NodeModel.ts

export interface NodeConfig {
  id: string;
  type: string;                        // 节点类型（动态，不限定）
  title: string;
  description?: string;
  position?: { x: number; y: number };

  // 🔥 核心：所有配置都是动态的
  config: {
    // 配置表单定义（动态）
    formSchema?: FormFieldSchema[];

    // 配置表单的值（动态）
    formValues?: Record<string, any>;

    // 预定义的常用配置（可选）

    permissions?: PermissionConfig;
    conditions?: ConditionConfig[];
    timeout?: TimeoutConfig;

    // 自定义配置（完全开放）
    [key: string]: any;
  };

  // 样式配置（动态）
  style?: NodeStyleConfig;

  // 子节点
  children?: string[];
}

// 表单字段 Schema（动态定义表单）
export interface FormFieldSchema {
  name: string;                        // 字段名
  label: string;                       // 字段标签
  type: FormFieldType;                 // 字段类型
  component?: string;                  // 自定义组件名（可选）

  required?: boolean;                  // 是否必填
  placeholder?: string;                // 占位符
  disabled?: boolean;                  // 是否禁用

    value: any;
    disabled?: boolean;

  validation?: ValidationRule[];       // 验证规则

### 2. 配置引擎（核心）

```typescript
// packages/core/src/engine/ConfigEngine.ts

export class ConfigEngine {
  /**
   * 加载配置
   */
  loadConfig(config: FlowConfig): FlowModel {
    // 验证配置
    this.validateConfig(config);

    // 创建流程模型
    return new FlowModel(config);
  }

  /**
   * 解析节点配置
   */
  parseNodeConfig(nodeConfig: NodeConfig): ParsedNodeConfig {
    // 解析 formSchema 生成表单
    const formFields = this.parseFormSchema(nodeConfig.config.formSchema || []);

    // 解析条件表达式
    const conditions = this.parseConditions(nodeConfig.config.conditions || []);

    return {
      ...nodeConfig,
      parsedForm: formFields,
      parsedConditions: conditions
    };
  }

  /**
   * 解析表单 Schema
   */
  parseFormSchema(schema: FormFieldSchema[]): ParsedFormField[] {
    return schema.map(field => ({
      ...field,
      // 解析 visible 表达式
      isVisible: typeof field.visible === 'string'
        ? this.evaluateExpression(field.visible)
        : field.visible !== false,
      // 解析默认值
      value: field.defaultValue
    }));
  }

  /**
   * 表达式求值（简单版）
   */
  evaluateExpression(expr: string, context?: Record<string, any>): boolean {
    // 支持简单表达式，如：amount > 10000
    // 可以用 eval 或者自己实现表达式解析器
    try {
      const func = new Function(...Object.keys(context || {}), `return ${expr}`);
      return func(...Object.values(context || {}));
    } catch {
      return false;
    }
  }

  /**
   * 验证配置
   */
  validateConfig(config: FlowConfig): ValidationResult {
    const errors: string[] = [];

    // 验证节点
    config.nodes.forEach(node => {
      if (!node.id) errors.push('节点缺少 id');
      if (!node.type) errors.push(`节点 ${node.id} 缺少 type`);
    });

    // 验证连线
    config.edges.forEach(edge => {
      const sourceExists = config.nodes.some(n => n.id === edge.source);
      const targetExists = config.nodes.some(n => n.id === edge.target);
      if (!sourceExists) errors.push(`连线 ${edge.id} 的源节点不存在`);
      if (!targetExists) errors.push(`连线 ${edge.id} 的目标节点不存在`);
    });

    return {
      valid: errors.length === 0,
      errors
    };
  }
}
```

### 动态表单组件（核心）
```tsx
// packages/react/src/components/config/DynamicForm.tsx

import React from 'react';
import { FormFieldSchema } from '@yourname/flow-core';

export interface DynamicFormProps {
  // 表单 Schema（从配置中来）
  schema: FormFieldSchema[];

  // 表单值
  values: Record<string, any>;

  // 值变化回调
  onChange: (values: Record<string, any>) => void;

  // 自定义组件映射
  components?: Record<string, React.ComponentType<any>>;
}

export const DynamicForm: React.FC<DynamicFormProps> = ({
  schema,
  values,
  onChange,
  components = {}
}) => {
  const renderField = (field: FormFieldSchema) => {
    // 处理可见性
    const isVisible = typeof field.visible === 'string'
      ? evaluateExpression(field.visible, values)
      : field.visible !== false;

    if (!isVisible) return null;

    // 如果指定了自定义组件
    if (field.component && components[field.component]) {
      const CustomComponent = components[field.component];
      return (
        <CustomComponent
          {...field.props}
          value={values[field.name]}
          onChange={(value: any) => onChange({ ...values, [field.name]: value })}
        />
      );
    }

    // 使用内置组件
    switch (field.type) {
      case 'input':
        return (
          <Input
            placeholder={field.placeholder}
            value={values[field.name]}
            onChange={(e) => onChange({ ...values, [field.name]: e.target.value })}
            disabled={field.disabled}
          />
        );

      case 'textarea':
        return (
          <TextArea
            placeholder={field.placeholder}
            value={values[field.name]}
            onChange={(e) => onChange({ ...values, [field.name]: e.target.value })}
            disabled={field.disabled}
            rows={field.props?.rows || 3}
          />
        );

      case 'number':
        return (
          <InputNumber
            placeholder={field.placeholder}
            value={values[field.name]}
            onChange={(value) => onChange({ ...values, [field.name]: value })}
            disabled={field.disabled}
            {...field.props}
          />
        );

      case 'select':
        return (
          <Select
            placeholder={field.placeholder}
            value={values[field.name]}
            onChange={(value) => onChange({ ...values, [field.name]: value })}
            disabled={field.disabled}
            {...field.props}
          >
            {field.options?.map(opt => (
              <Select.Option key={opt.value} value={opt.value} disabled={opt.disabled}>
                {opt.label}
              </Select.Option>
            ))}
          </Select>
        );

      case 'radio':
        return (
          <Radio.Group
            value={values[field.name]}
            onChange={(e) => onChange({ ...values, [field.name]: e.target.value })}
            disabled={field.disabled}
          >
            {field.options?.map(opt => (
              <Radio key={opt.value} value={opt.value} disabled={opt.disabled}>
                {opt.label}
              </Radio>
            ))}
          </Radio.Group>
        );

      case 'checkbox':
        return (
          <Checkbox.Group
            value={values[field.name]}
            onChange={(value) => onChange({ ...values, [field.name]: value })}
            disabled={field.disabled}
          >
            {field.options?.map(opt => (
              <Checkbox key={opt.value} value={opt.value} disabled={opt.disabled}>
                {opt.label}
              </Checkbox>
            ))}
          </Checkbox.Group>
        );

      case 'switch':
        return (
          <Switch
            checked={values[field.name]}
            onChange={(checked) => onChange({ ...values, [field.name]: checked })}
            disabled={field.disabled}
          />
        );

      case 'date':
        return (
          <DatePicker
            placeholder={field.placeholder}
            value={values[field.name]}
            onChange={(date) => onChange({ ...values, [field.name]: date })}
            disabled={field.disabled}
            {...field.props}
          />
        );

      case 'dateRange':
        return (
          <DatePicker.RangePicker
            placeholder={[field.placeholder || '开始日期', '结束日期']}
            value={values[field.name]}
            onChange={(dates) => onChange({ ...values, [field.name]: dates })}
            disabled={field.disabled}
            {...field.props}
          />
        );

      // ... 其他类型

      default:
        return <div>未知字段类型: {field.type}</div>;
    }
  };

  return (
    <Form layout="vertical">
      {schema.map(field => (
        <Form.Item
          key={field.name}
          label={field.label}
          required={field.required}
        >
          {renderField(field)}
        </Form.Item>
      ))}
    </Form>
  );
};
```

      <a-form-item :label="field.label" :required="field.required">

        <component
          v-if="field.component && components[field.component]"
          :is="components[field.component]"
          v-bind="field.props"
          :value="values[field.name]"
          @update:value="handleChange(field.name, $event)"

        <!-- 内置组件 -->
        <template v-else>

          <a-input
            v-if="field.type === 'input'"
            v-model:value="values[field.name]"

          <a-textarea
            v-else-if="field.type === 'textarea'"
            v-model:value="values[field.name]"
            :placeholder="field.placeholder"
            :disabled="field.disabled"
            :rows="field.props?.rows || 3"
          />

          <!-- 数字 -->
          <a-input-number
            v-else-if="field.type === 'number'"
            v-model:value="values[field.name]"
            :placeholder="field.placeholder"
            :disabled="field.disabled"
            v-bind="field.props"
          />

          <!-- 下拉选择 -->
          <a-select
            v-else-if="field.type === 'select'"
            v-model:value="values[field.name]"
            :placeholder="field.placeholder"
            :disabled="field.disabled"
            v-bind="field.props"
          >
            <a-select-option
              v-for="opt in field.options"
              :key="opt.value"
              :value="opt.value"
              :disabled="opt.disabled"
            >
              {{ opt.label }}
            </a-select-option>
          </a-select>

          <!-- 单选框 -->
          <a-radio-group
            v-else-if="field.type === 'radio'"
            v-model:value="values[field.name]"
            :disabled="field.disabled"
          >
            <a-radio
              v-for="opt in field.options"
              :key="opt.value"
              :value="opt.value"
              :disabled="opt.disabled"
            >
              {{ opt.label }}
            </a-radio>
          </a-radio-group>

          <!-- 多选框 -->
          <a-checkbox-group
            v-else-if="field.type === 'checkbox'"
            v-model:value="values[field.name]"
            :disabled="field.disabled"
          >
            <a-checkbox
              v-for="opt in field.options"
              :key="opt.value"
              :value="opt.value"
              :disabled="opt.disabled"
            >
              {{ opt.label }}
            </a-checkbox>
          </a-checkbox-group>

          <!-- 日期 -->
          <a-date-picker
            v-else-if="field.type === 'date'"
            v-model:value="values[field.name]"
            :placeholder="field.placeholder"
            :disabled="field.disabled"
            v-bind="field.props"
          />

          <!-- 日期区间 -->
          <a-range-picker
            v-else-if="field.type === 'dateRange'"
            v-model:value="values[field.name]"
            :placeholder="[field.placeholder || '开始日期', '结束日期']"
            :disabled="field.disabled"
            v-bind="field.props"
          />

          <!-- 其他类型... -->
        </template>
      </a-form-item>
    </template>

<script setup lang="ts">
import { computed } from 'vue';
import { FormFieldSchema } from '@yourname/flow-core';

interface Props {
  schema: FormFieldSchema[];
  values: Record<string, any>;
  components?: Record<string, any>;
}

const props = defineProps<Props>();
const emit = defineEmits<{
  (e: 'change', values: Record<string, any>): void;
}>();

// 计算可见字段
const visibleFields = computed(() => {
  return props.schema.filter(field => {
    if (typeof field.visible === 'string') {
      // 表达式求值
      return evaluateExpression(field.visible, props.values);
    }
    return field.visible !== false;
  });
});

const handleChange = (name: string, value: any) => {
  emit('change', { ...props.values, [name]: value });
};

// 表达式求值
const evaluateExpression = (expr: string, context: Record<string, any>): boolean => {
  try {
    const func = new Function(...Object.keys(context), `return ${expr}`);
    return func(...Object.values(context));
  } catch {
    return false;
  }
};
</script>
