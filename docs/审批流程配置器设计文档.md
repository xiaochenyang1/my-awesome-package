## 🎯 产品定位

**一个可视化的审批流程配置器，支持分支条件、节点配置、权限管理**

### 核心特点
- ✅ 支持分支流程（通过/驳回/自定义条件）
- ✅ 支持条件节点（if/else）
- ✅ 节点可配置详细信息（权限、表单项等）
- ✅ 拖拽操作，可视化编排
- ✅ 实时预览效果

## 💻 数据结构设计

### 节点数据结构
```typescript
// 节点类型
type NodeType =
  | 'start'       // 开始节点（提交申请）
  | 'approval'    // 审批节点
  | 'condition'   // 条件节点
  | 'end';        // 结束节点

// 节点状态
type NodeStatus = 'pending' | 'active' | 'completed' | 'rejected';

// 节点定义
interface FlowNode {
  id: string;                    // 节点唯一ID
  type: NodeType;                // 节点类型
  title: string;                 // 节点标题
  description?: string;          // 节点描述

  // 位置信息（用于自由布局）
  position?: {
    x: number;
    y: number;
  };

  // 节点配置
  config: {
    // 审批人配置
    approvers?: {
      type: 'user' | 'role' | 'dept';  // 审批人类型
      mode: 'single' | 'multiple' | 'any';  // 审批模式（单人/会签/或签）
      users?: string[];                // 指定用户
      roles?: string[];                // 指定角色
      depts?: string[];                // 指定部门
      autoPass?: boolean;              // 自动通过（发起人在审批人中）
    };

    // 表单权限配置
    formPermissions?: {
      [fieldName: string]: {
        visible: boolean;     // 是否可见
        editable: boolean;    // 是否可编辑
        required: boolean;    // 是否必填
      };
    };

    // 条件配置（条件节点）
    conditions?: Array<{
      id: string;
      expression: string;           // 条件表达式，如 "amount > 10000"
      operator: 'and' | 'or';       // 多条件组合方式
      targetNodeId: string;         // 满足条件跳转的节点
    }>;

    // 超时设置
    timeout?: {
      enabled: boolean;
      duration: number;             // 超时时长（小时）
      action: 'auto_pass' | 'auto_reject' | 'remind';  // 超时动作
    };

    // 自定义字段
    customFields?: Array<{
      name: string;
      label: string;
      type: 'input' | 'select' | 'checkbox' | 'radio' | 'date' | 'dateRange' | 'textarea';
      value?: any;
      options?: Array<{ label: string; value: any }>;  // 下拉/单选/多选的选项
      required?: boolean;
      placeholder?: string;
    }>;
  };

  // 连线信息
  children?: string[];           // 子节点ID列表（支持多个分支）
}

// 流程定义
interface FlowData {
  id: string;                    // 流程ID
  name: string;                  // 流程名称
  version: string;               // 版本号
  nodes: FlowNode[];             // 节点列表
  edges: FlowEdge[];             // 连线列表
  startNodeId: string;           // 开始节点ID
}

// 连线定义
interface FlowEdge {
  id: string;                    // 连线ID
  source: string;                // 源节点ID
  target: string;                // 目标节点ID
  label?: string;                // 连线标签（如"通过"、"驳回"）
  condition?: string;            // 条件表达式
  style?: {
    type?: 'solid' | 'dashed';
    color?: string;
  };
}
```

## 🛠️ 核心功能实现

### 1. 添加分支节点
```typescript
function addBranch(parentNodeId: string, branchType: 'pass' | 'reject' | 'custom') {
  const newNode: FlowNode = {
    id: generateId(),
    type: 'approval',
    title: branchType === 'pass' ? '审批通过' : branchType === 'reject' ? '审批驳回' : '新分支',
    config: {
      approvers: {
        type: 'user',
        mode: 'single',
        users: []
      }
    },
    children: []
  };

  const newEdge: FlowEdge = {
    id: generateId(),
    source: parentNodeId,
    target: newNode.id,
    label: branchType === 'pass' ? '通过' : branchType === 'reject' ? '驳回' : ''
  };

  // 更新父节点的 children
  const updatedNodes = nodes.map(node => {
    if (node.id === parentNodeId) {
      return {
        ...node,
        children: [...(node.children || []), newNode.id]
      };
    }
    return node;
  });

  setNodes([...updatedNodes, newNode]);
  setEdges([...edges, newEdge]);
}
```

### 2. 删除节点（重新连接）
```typescript
function deleteNode(nodeId: string) {
  // 找到父节点
  const parentNode = nodes.find(n => n.children?.includes(nodeId));

  // 找到当前节点的子节点
  const currentNode = nodes.find(n => n.id === nodeId);
  const childrenIds = currentNode?.children || [];

  if (parentNode) {
    // 将当前节点的子节点连接到父节点
    const updatedParent = {
      ...parentNode,
      children: [
        ...(parentNode.children?.filter(id => id !== nodeId) || []),
        ...childrenIds
      ]
    };

    setNodes(nodes.map(n => n.id === parentNode.id ? updatedParent : n).filter(n => n.id !== nodeId));

    // 更新连线
    setEdges(edges.filter(e => e.source !== nodeId && e.target !== nodeId));
  }
}
```

### 3. 节点配置保存
```typescript
function saveNodeConfig(nodeId: string, config: Partial<FlowNode>) {
  setNodes(nodes.map(node => {
    if (node.id === nodeId) {
      return { ...node, ...config };
    }
    return node;
  }));
}
```

## 💻 核心代码示例

### 使用 ReactFlow 实现
```tsx
import ReactFlow, {
  Node,
  Edge,
  Controls,
  Background,
  MiniMap,
  useNodesState,
  useEdgesState
} from 'reactflow';
import 'reactflow/dist/style.css';

// 自定义节点类型
const nodeTypes = {
  start: StartNode,
  approval: ApprovalNode,
  condition: ConditionNode,
  end: EndNode
};

function FlowDesigner() {
  const [nodes, setNodes, onNodesChange] = useNodesState(initialNodes);
  const [edges, setEdges, onEdgesChange] = useEdgesState(initialEdges);
  const [selectedNode, setSelectedNode] = useState<Node | null>(null);
  const [configModalVisible, setConfigModalVisible] = useState(false);

  // 节点点击
  const onNodeClick = (event: React.MouseEvent, node: Node) => {
    setSelectedNode(node);
  };

  // 节点双击 - 打开配置弹窗
  const onNodeDoubleClick = (event: React.MouseEvent, node: Node) => {
    setSelectedNode(node);
    setConfigModalVisible(true);
  };

  // 添加节点
  const onAddNode = (parentId: string, type: NodeType) => {
    const newNode: Node = {
      id: generateId(),
      type: type,
      position: { x: 0, y: 0 },  // 会被自动布局重新计算
      data: {
        title: '新节点',
        config: {}
      }
    };

    const newEdge: Edge = {
      id: generateId(),
      source: parentId,
      target: newNode.id,
      type: 'smoothstep',
      animated: true
    };

    setNodes([...nodes, newNode]);
    setEdges([...edges, newEdge]);

    // 自动布局
    autoLayout();
  };

  // 自动布局
  const autoLayout = () => {
    const layoutedElements = getLayoutedElements(nodes, edges);
    setNodes(layoutedElements.nodes);
  };

  return (
    <div style={{ width: '100%', height: '100vh' }}>
      <ReactFlow
        nodes={nodes}
        edges={edges}
        onNodesChange={onNodesChange}
        onEdgesChange={onEdgesChange}
        onNodeClick={onNodeClick}
        onNodeDoubleClick={onNodeDoubleClick}
        nodeTypes={nodeTypes}
        fitView
      >
        <Background />
        <Controls />
        <MiniMap />
      </ReactFlow>

      {/* 节点配置弹窗 */}
      <NodeConfigModal
        visible={configModalVisible}
        node={selectedNode}
        onOk={(updatedNode) => {
          setNodes(nodes.map(n => n.id === updatedNode.id ? updatedNode : n));
          setConfigModalVisible(false);
        }}
        onCancel={() => setConfigModalVisible(false)}
      />
    </div>
  );
}
```
